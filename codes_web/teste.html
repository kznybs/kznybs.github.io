Questão 1: Pódio da Gincana

algoritmo "Questao1"
var
   nome, nome1, nome2, nome3: caractere
   n1, n2, n3, media: real
   nota1, nota2, nota3: real
   cont: inteiro
inicio
   nota1 <- 0
   nota2 <- 0
   nota3 <- 0

   para cont de 1 ate 5 faca
      escreval("Digite o nome da equipe:")
      leia(nome)
      escreval("Nota do jurado 1:")
      leia(n1)
      escreval("Nota do jurado 2:")
      leia(n2)
      escreval("Nota do jurado 3:")
      leia(n3)

      media <- (n1 + n2 + n3) / 3

      se media > nota1 entao
         nota3 <- nota2
         nome3 <- nome2
         nota2 <- nota1
         nome2 <- nome1
         nota1 <- media
         nome1 <- nome
      senao
         se media > nota2 entao
            nota3 <- nota2
            nome3 <- nome2
            nota2 <- media
            nome2 <- nome
         senao
            se media > nota3 entao
               nota3 <- media
               nome3 <- nome
            fimse
         fimse
      fimse
   fimpara

   escreval("1o lugar: ", nome1, " com nota ", nota1)
   escreval("2o lugar: ", nome2, " com nota ", nota2)
   escreval("3o lugar: ", nome3, " com nota ", nota3)
fimalgoritmo

Questão 2: Análise de Notas da Turma

algoritmo "Questao2"
var
   nome, nome_maior: caractere
   nota, soma, media, maior_nota: real
   total, aprovados, recup, reprovados: inteiro
inicio
   soma <- 0
   total <- 0
   aprovados <- 0
   recup <- 0
   reprovados <- 0
   maior_nota <- 0

   escreval("Digite o nome do aluno:")
   leia(nome)
   escreval("Digite a nota (ou -1 para parar):")
   leia(nota)

   enquanto nota <> -1 faca
      soma <- soma + nota
      total <- total + 1

      se nota >= 7 entao
         aprovados <- aprovados + 1
      senao
         se nota >= 5 entao
            recup <- recup + 1
         senao
            reprovados <- reprovados + 1
         fimse
      fimse

      se nota > maior_nota entao
         maior_nota <- nota
         nome_maior <- nome
      fimse

      escreval("Digite o nome do proximo aluno:")
      leia(nome)
      escreval("Digite a nota (ou -1 para parar):")
      leia(nota)
   fimenquanto

   se total > 0 entao
      media <- soma / total
      escreval("Media geral da turma: ", media)
      escreval("Quantidade total de alunos: ", total)
      escreval("Quantidade de aprovados: ", aprovados)
      escreval("Quantidade em recuperacao: ", recup)
      escreval("Quantidade de reprovados: ", reprovados)
      escreval("Nome do aluno com maior nota: ", nome_maior)
      escreval("Maior nota registrada: ", maior_nota)
   fimse
fimalgoritmo
